  that directory as `INFERATLAS_DATA_DIR`. Optional `--data-dir` exports replace

  `Ascend 替换 — 当前参数下无已审阅说明` and its class/source links. The inactive
  explanation's badges and annotation id are cleared; returning to BF16 restores
  the original note. Unit and browser assertions cover this behavior.

  `o_proj` and `MLPRowParallelOp` for `down_proj`, including the latter's token
  reduce-scatter. Both implementations and the selector were already hashed

  extract. A missing match or any differing boolean blocks binding, even when
  that boolean is unused by the predicate and all predicate results are true.

## Decisions and contract changes

The orchestrator-approved pinned-commit defaults rule is retained. Missing or
unknown facts for other revision pairs still triage rather than selecting a lane.
The `conditional_override` discriminator remains compatible; its optional `reason`
adds the blocking unannotated case. `AscendTag` gains optional `applies_when` plus
fixed boolean `context`; `ExtractOverride` gains optional `attrs`. Hardware
presets gain nonnegative overhead estimates defaulting to zero where unknown.
Pydantic, JSON Schema and generated TypeScript are regenerated together.

Golden updates reflect annotations/runtime contexts, EP choices and catalog memory
deductions; weight accounting is unchanged. TP=1 resident parameter totals remain
16,381,470,720 bytes for dense and 61,064,245,248 bytes for MoE. Both TP=1 and TP=4
weights match the real extracted parameters in each verification report.

No deviation from the requested local build/data contracts remains. Retained
Actions snapshots use immutable snapshot ids as index version ids, preserving
history even when multiple builds share the upstream display pair. This is the
existing retained-state contract, separate from the CLI's catalog-pair index.

import importlib.util
import json
import os
from pathlib import Path
import shutil

spec = importlib.util.spec_from_file_location('pipeline', '.github/actions/pipeline.py')
pipeline = importlib.util.module_from_spec(spec)
spec.loader.exec_module(pipeline)
pipeline.CACHE = Path('/tmp/i1-retained-render')
shutil.rmtree(pipeline.CACHE, ignore_errors=True)
paths = sorted(Path('.cache/build').glob('*/*/ir.json'))
snapshot = json.loads(paths[0].read_text())['snapshot']
pair = {name: snapshot[name]['commit'] for name in ('vllm', 'ascend')}
os.environ['SNAPSHOT_ID'] = pipeline.snapshot_id(pair)
for path in paths:
    ir = json.loads(path.read_text())
    dest = pipeline.CACHE / 'state/data' / os.environ['SNAPSHOT_ID'] / path.parent.name / ir['snapshot']['device']
    dest.mkdir(parents=True)
    for name in ('ir.json', 'view.json', 'report.json'):
        shutil.copyfile(path.with_name(name), dest / name)
pipeline.render()
outputs = sorted((Path('dist/single') / os.environ['SNAPSHOT_ID']).glob('*.html'))
assert len(outputs) == 4
assert Path('dist/site/index.html').is_file()
for path in outputs:
    print(path, path.stat().st_size, 'bytes')
print('Retained real matrix and all four release exports rendered successfully.')

from pathlib import Path
import json

def read(path):
    return json.loads(Path(path).read_text())


index = read('web/dist/data/index.json')
cells = [c for c in index['cells'] if c['status'] == 'ok']

]

print('Real matrix:', sorted((c['model'], c['device']) for c in cells))
print('model | device | extract.json | extract-tp4.json | ir.json | view.json | report.json | triage.json (bytes)')
for path in sorted(Path('.cache/build').glob('*/*/ir.json')):
    folder = path.parent
    ir = read(path)
    extracts = [read(folder / n) for n in ('extract.json', 'extract-tp4.json')]
    assert not any(e['fixture'] for e in extracts)
    assert not read(folder / 'triage.json')['items']
    report = read(folder / 'report.json')
    assert not report['golden_diff'] and not report['source_link_failures']
    assert not any(c['status'] == 'fail' for c in report['checks'])
    assert next(c for c in report['checks'] if c['id'] == 'provenance')['status'] == 'pass'
    replacements = {m['ascend']['class'] for e in extracts for m in e['modules'] if m.get('ascend')}
    annotated = [n for n in ir['nodes'] if n.get('ascend')]
    covered = {s['symbol'] for n in annotated for s in n['ascend']['src']}
    assert replacements <= covered
    assert all(n['ascend']['applies_when'] and 'dtype_w' not in n['ascend']['context'] for n in annotated)
    names = ('extract.json', 'extract-tp4.json', 'ir.json', 'view.json', 'report.json', 'triage.json')
    print(folder.name, ir['snapshot']['device'], *( (folder / n).stat().st_size for n in names), sep=' | ')
    print('  replacement implementations:', len(replacements), '; annotated IR nodes:', len(annotated))
    print('  view extents:', [(v['id'], v['width'], v['height']) for v in read(folder / 'view.json')['views']])
for device in ('A2', 'H100'):
    hardware = next(p for p in read('web/dist/data/hardware.json') if p['device'] == device)
    assert hardware['non_torch_bytes'] > 0 and hardware['reserved_bytes'] > 0
    print(device, 'hardware bytes:', hardware['non_torch_bytes'], hardware['reserved_bytes'])
for model in ('qwen3-8b', 'qwen3-30b-a3b'):
    path = Path(f'dist/inferatlas-{model}.html')
    payload = json.loads(re.search(r'window\.__INFERATLAS__=(.*?);</script>', path.read_text()).group(1))
    assert [m['id'] for m in payload['index']['models']] == [model]

for name in ('web/dist/data/index.json', 'web/dist/data/hardware.json', 'web/dist/index.html', 'web/dist-single/index.html'):
    print(name, Path(name).stat().st_size, 'bytes')
